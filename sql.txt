-- INFRAESTRUTURA PARA GESTAO DE DADOS - TRABALHO 2

CREATE TABLE T2_USUARIOS (
    usuario_id NUMBER(5) CONSTRAINT PK_T2_USUARIOS PRIMARY KEY,
    nome VARCHAR2(80) NOT NULL,
    email VARCHAR2(120) NOT NULL,
    CONSTRAINT UQ_T2_USUARIOS_EMAIL UNIQUE (email)
);

CREATE TABLE T2_SALAS (
    sala_id NUMBER(5) CONSTRAINT PK_T2_SALAS PRIMARY KEY,
    nome_sala VARCHAR2(60) NOT NULL,
    capacidade NUMBER(4) NOT NULL,
    CONSTRAINT CK_T2_SALAS_CAP CHECK (capacidade > 0)
);

CREATE TABLE T2_RESERVAS (
    reserva_id NUMBER(6) CONSTRAINT PK_T2_RESERVAS PRIMARY KEY,
    sala_id NUMBER(5) NOT NULL,
    usuario_id NUMBER(5) NOT NULL,
    inicio TIMESTAMP NOT NULL,
    fim TIMESTAMP NOT NULL,
    status VARCHAR2(12) DEFAULT 'ATIVA' NOT NULL,
    CONSTRAINT FK_T2_RES_SALA FOREIGN KEY (sala_id) REFERENCES T2_SALAS(sala_id),
    CONSTRAINT FK_T2_RES_USUARIO FOREIGN KEY (usuario_id) REFERENCES T2_USUARIOS(usuario_id),
    CONSTRAINT CK_T2_RES_PERIODO CHECK (fim > inicio),
    CONSTRAINT CK_T2_RES_STATUS CHECK (status IN ('ATIVA','CANCELADA'))
);
